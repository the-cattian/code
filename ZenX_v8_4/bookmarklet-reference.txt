function getAudioContext() {
  try {
    if (!audioCtx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) audioCtx = new AudioCtx();
    }
    if (audioCtx && audioCtx.state === 'suspended') audioCtx.resume().catch(() => {});
    return audioCtx;
  } catch(e) {
    return null;
  }
}

function playTone(freq, duration, type = 'sine', volume = 0.22) {
  if (isMuted) return;
  try {
    const ctx = getAudioContext();
    if (!ctx) return;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, ctx.currentTime);
    gain.gain.setValueAtTime(volume, ctx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + duration);
    osc.connect(gain);
    gain.connect(ctx.destination);
    osc.start();
    osc.stop(ctx.currentTime + duration);
  } catch(e) {}
}

function isRelativeTimeText(str) {
  if (!str || typeof str !== 'string') return false;
  const s = str.trim().toLowerCase();
  return /\b(?:less than|under)\s+(?:a|1)\s+min(?:ute)?s?\b/i.test(s) ||
         /\b(?:just now|right now|a moment ago|a few seconds? ago)\b/i.test(s) ||
         /\b\d+\s*(?:sec(?:ond)?s?|min(?:ute)?s?|hours?|hrs?)\s+ago\b/i.test(s) ||
         /\b(?:a|one)\s+(?:min(?:ute)?|second|hour)\s+ago\b/i.test(s);
}

function parseRelativeElapsedSeconds(rawStr) {
  if (!rawStr || typeof rawStr !== 'string') return null;
  const s = rawStr.trim().toLowerCase();
  if (!s) return null;
  if (/\b(?:less than|under)\s+(?:a|1)\s+min(?:ute)?s?\b/i.test(s)) return 0;
  if (/\b(?:just now|right now|a moment ago|a few seconds? ago)\b/i.test(s)) return 0;
  let m = s.match(/\b(\d+|a|one)\s*(?:seconds?|secs?)\s+ago\b/i);
  if (m) {
    return (m[1] === 'a' || m[1] === 'one') ? 1 : parseInt(m[1], 10);
  }
  m = s.match(/\b(\d+|a|one)\s*(?:minutes?|mins?)\s+ago\b/i);
  if (m) {
    const mins = (m[1] === 'a' || m[1] === 'one') ? 1 : parseInt(m[1], 10);
    return mins * 60;
  }
  m = s.match(/\b(\d+|a|one)\s*(?:hours?|hrs?)\s+ago\b/i);
  if (m) {
    const hrs = (m[1] === 'a' || m[1] === 'one') ? 1 : parseInt(m[1], 10);
    return hrs * 3600;
  }
  return null;
}

  if (direct && isRelativeTimeText(direct)) return direct.trim();
  const article = box.closest('article[data-test-id="omni-log-comment-item"]');
  const scope = article || box.parentElement || box;
  const timeEl = scope.querySelector('[data-test-id="timestamp-relative"], [data-test-id*="timestamp"], time');
  if (timeEl) {
    const txt = (timeEl.textContent || '').trim();
    if (isRelativeTimeText(txt)) return txt;
    const aria = (timeEl.getAttribute('aria-label') || '').trim();
    if (isRelativeTimeText(aria)) return aria;
  }
  const allEls = scope.querySelectorAll('*');
  for (const el of allEls) {
    if (el.children.length === 0) {
      const txt = (el.textContent || '').trim();
      if (isRelativeTimeText(txt)) return txt;
    }
  }
  return '';
}

function getAgentMessages(container) {

function evaluateAgentSLA() {
  const ticketInfo = getActiveTicketDetails();
  if (!isTicketOpen(ticketInfo.container)) {
    return {
      isOpen: false,
      ticketId: ticketInfo.ticketId,
      ticketKey: ticketInfo.ticketKey,
      displayBadge: ticketInfo.displayBadge,
      nativeSla: ticketInfo.nativeSla,
      durationSec: 0,
      remainingSec: 0,
      remainingMs: 0,
      rule: 'idle',
      label: 'Idle',
      msgId: ''
    };
  }

  const tState = getTicketState(ticketInfo.ticketKey);
  tState.ticketId = ticketInfo.ticketId;
  tState.displayBadge = ticketInfo.displayBadge;
  tState.nativeSla = ticketInfo.nativeSla;

  const agentMsgs = getAgentMessages(ticketInfo.container);
  const scope = ticketInfo.container || document;
  const allArticles = Array.from(scope.querySelectorAll('article[data-test-id="omni-log-comment-item"]'));

  if (agentMsgs.length === 0) {
    const durationSec = tState.manualDurationOverride > 0 ? tState.manualDurationOverride : 60;
    const firstReplyKey = 'first_reply_' + ticketInfo.ticketKey;
    if (tState.msgId !== firstReplyKey) {
      tState.msgId = firstReplyKey;
      tState.rawTimeText = '';
      tState.hasBeeped1Min = false;
      tState.macroDismissedId = '';
      let custElapsed = 0;
      if (allArticles.length) {
        const lastArticle = allArticles[allArticles.length - 1];
        const timeEl = lastArticle.querySelector('[data-test-id="timestamp-relative"], time');
        const relStr = timeEl ? (timeEl.textContent || '').trim() : '';
        const parsed = parseRelativeElapsedSeconds(relStr);
        if (parsed !== null) custElapsed = parsed;
      }
      tState.baseElapsedSec = (tState.manualResetTime > 0 && Date.now() - tState.manualResetTime < 30000) ? 0 : custElapsed;
      tState.anchorTime = Date.now();
      tState.manualResetTime = 0;
    }
    const localSec = Math.max(0, Math.floor((Date.now() - tState.anchorTime) / 1000));
    const totalElapsed = tState.baseElapsedSec + localSec;
    const remainingSec = Math.max(0, durationSec - totalElapsed);
    return {
      isOpen: true,
      ticketId: ticketInfo.ticketId,
      ticketKey: ticketInfo.ticketKey,
      displayBadge: ticketInfo.displayBadge,
      nativeSla: ticketInfo.nativeSla,
      durationSec,
      remainingSec,
      remainingMs: remainingSec * 1000,
      rule: 'first_reply',
      label: '1m First Reply',
      msgId: firstReplyKey
    };
  }

  const latestAgent = agentMsgs[agentMsgs.length - 1];
  const lastAgentArticle = latestAgent.element ? latestAgent.element.closest('article[data-test-id="omni-log-comment-item"]') : null;
  const lastAgentIndex = lastAgentArticle ? allArticles.indexOf(lastAgentArticle) : -1;

  let customerRepliedAfter = false;
  let latestCustArticle = null;
  if (lastAgentIndex !== -1 && lastAgentIndex < allArticles.length - 1) {
    for (let i = allArticles.length - 1; i > lastAgentIndex; i--) {
      const art = allArticles[i];
      const role = (art.getAttribute('data-chat-role') || '').toLowerCase();
      const isInternal = !!art.querySelector('[data-test-id="omni-log-internal-note-tag"], [type="internal"]');
      if (!isInternal && role !== 'agent' && role !== 'internal') {
        customerRepliedAfter = true;
        latestCustArticle = art;
        break;
      }
    }
  }

  let durationSec = 120;
  let rule = 'general';
  let label = '2m Response';
  let msgIdentifier = '';
  let currentRawTime = '';

  if (customerRepliedAfter && latestCustArticle) {
    durationSec = 120;
    rule = 'customer_reply';
    label = '2m Response';
    const timeEl = latestCustArticle.querySelector('[data-test-id="timestamp-relative"], time');
    currentRawTime = timeEl ? (timeEl.textContent || '').trim() : '';
    msgIdentifier = 'cust_reply_' + (latestCustArticle.getAttribute('data-event-id') || latestCustArticle.id || latestCustArticle.textContent.slice(0, 30));
  } else {
    const isHold = HOLD_REGEX.test(latestAgent.text);
    const isAssistance = !isHold && ASSISTANCE_REGEX.test(latestAgent.text);
    durationSec = isHold ? 300 : (isAssistance ? 60 : 120);
    rule = isHold ? 'hold' : (isAssistance ? 'assistance' : 'general');
    label = isHold ? '5m Hold' : (isAssistance ? '1m Assistance' : '2m Response');
    msgIdentifier = (latestAgent.id || '') + '_agent_' + latestAgent.text.slice(0, 40);
    currentRawTime = (latestAgent.rawTime || '').trim();
  }

  if (tState.manualDurationOverride > 0) {
    durationSec = tState.manualDurationOverride;
    label = `${Math.round(durationSec / 60)}m Manual`;
  }

  const parsedElapsed = parseRelativeElapsedSeconds(currentRawTime);

  if (msgIdentifier !== tState.msgId) {
    tState.msgId = msgIdentifier;
    tState.rawTimeText = currentRawTime;
    tState.hasBeeped1Min = false;
    tState.macroDismissedId = '';
    if (tState.manualResetTime > 0 && Date.now() - tState.manualResetTime < 30000) {
      tState.baseElapsedSec = 0;
    } else if (parsedElapsed !== null) {
      tState.baseElapsedSec = parsedElapsed;
    } else {
      tState.baseElapsedSec = 0;
    }
    tState.anchorTime = Date.now();
    tState.manualResetTime = 0;
  } else {
    if (currentRawTime && currentRawTime !== tState.rawTimeText && parsedElapsed !== null) {
      tState.rawTimeText = currentRawTime;
      if (parsedElapsed > tState.baseElapsedSec) {
        tState.baseElapsedSec = parsedElapsed;
        tState.anchorTime = Date.now();
      }
    }
  }

  const localSec = Math.max(0, Math.floor((Date.now() - tState.anchorTime) / 1000));
  const totalElapsed = tState.baseElapsedSec + localSec;
  const remainingSec = Math.max(0, durationSec - totalElapsed);

  return {
    isOpen: true,
    ticketId: ticketInfo.ticketId,
    ticketKey: ticketInfo.ticketKey,
    displayBadge: ticketInfo.displayBadge,
    nativeSla: ticketInfo.nativeSla,
    durationSec,
    remainingSec,
    remainingMs: remainingSec * 1000,
    rule,
    label,
    msgId: msgIdentifier
  };
}
function findZendeskComposer(container) {
  const selectors = [
    '[data-test-id="omni-composer-text-input"]',
    '[data-test-id="omni-log-composer-editor"] [contenteditable="true"]',
    '[data-test-id="omni-log-composer-editor"]',
    '[data-test-id*="composer"] [contenteditable="true"]',
    '[data-test-id*="editor"] [contenteditable="true"]',
    'div.ck-editor__editable[contenteditable="true"]',
    'div[role="textbox"][contenteditable="true"]',
    'div.zendesk-editor--rich-text-container [contenteditable="true"]',
    'textarea[data-test-id*="composer"]',
    'textarea[data-test-id*="editor"]',
    'textarea[data-test-id*="text-input"]',
    'textarea.zendesk-editor--textarea',
    'textarea[name*="comment"]',
    '[data-test-id*="composer"] textarea',
    '[data-test-id*="editor"] textarea',
    'div[contenteditable="true"]'
  ];
  if (container) {
    for (const sel of selectors) {
      try {
        const els = container.querySelectorAll(sel);
        for (const el of els) {
          const visible = el.offsetParent !== null || el.getClientRects().length > 0;
          const writable = el.tagName === 'TEXTAREA' || el.tagName === 'INPUT' || el.isContentEditable || el.getAttribute('contenteditable') === 'true' || el.getAttribute('role') === 'textbox';
          if (visible && writable && !el.disabled && !el.readOnly) return el;
        }
      } catch(e) {}
    }
  }
  for (const doc of collectDocs(document, [])) {
    try {
      for (const sel of selectors) {
        const els = doc.querySelectorAll(sel);
        for (const el of els) {
          const visible = el.offsetParent !== null || el.getClientRects().length > 0;
          const writable = el.tagName === 'TEXTAREA' || el.tagName === 'INPUT' || el.isContentEditable || el.getAttribute('contenteditable') === 'true' || el.getAttribute('role') === 'textbox';
          if (visible && writable && !el.disabled && !el.readOnly) return el;
        }
      }
    } catch(e) {}
  }
  return null;
}

function populateZendeskComposer(text) {
  const ticketInfo = getActiveTicketDetails();
  const el = findZendeskComposer(ticketInfo.container);
  if (!el) return false;
  const ownerDoc = el.ownerDocument || document;
  const view = (ownerDoc && ownerDoc.defaultView) ? ownerDoc.defaultView : window;
  try { el.focus(); } catch(e) {}
  if (el.tagName === 'TEXTAREA' || el.tagName === 'INPUT') {
    try {
      const proto = el.tagName === 'TEXTAREA' ? view.HTMLTextAreaElement.prototype : view.HTMLInputElement.prototype;
      const nativeSetter = Object.getOwnPropertyDescriptor(proto, 'value');
      if (nativeSetter && nativeSetter.set) {
        nativeSetter.set.call(el, text);
      } else {
        el.value = text;
      }
    } catch(e) {
      el.value = text;
    }
    try { el.dispatchEvent(new Event('input', { bubbles: true })); } catch(e) {}
    try { el.dispatchEvent(new Event('change', { bubbles: true })); } catch(e) {}
    return true;
  }
  if (el.isContentEditable || el.getAttribute('contenteditable') === 'true' || el.getAttribute('role') === 'textbox') {
    let ok = false;
    try {
      el.focus();
      const sel = view.getSelection ? view.getSelection() : window.getSelection();
      const range = ownerDoc.createRange();
      range.selectNodeContents(el);
      if (sel) {
        sel.removeAllRanges();
        sel.addRange(range);
      }
      ok = ownerDoc.execCommand('insertText', false, text);
    } catch(e) {}
    if (!ok || !el.textContent.includes(text.slice(0, 15))) {
      try {
        const paragraph = ownerDoc.createElement('p');
        paragraph.textContent = text;
        el.replaceChildren(paragraph);
      } catch(e) {
        el.textContent = text;
      }
      try {
        if (view.InputEvent) {
          el.dispatchEvent(new view.InputEvent('beforeinput', { bubbles: true, inputType: 'insertText', data: text }));
          el.dispatchEvent(new view.InputEvent('input', { bubbles: true, inputType: 'insertText', data: text }));
        }
      } catch(e) {
        try { el.dispatchEvent(new Event('input', { bubbles: true })); } catch(e2) {}
      }
      try { el.dispatchEvent(new Event('change', { bubbles: true })); } catch(e) {}
    }
    try {
      if (el.ckeditorInstance && typeof el.ckeditorInstance.setData === 'function') {
        el.ckeditorInstance.setData(`<p>${text}</p>`);
      }
    } catch(e) {}
    try {
      const sel = view.getSelection ? view.getSelection() : window.getSelection();
      if (sel) {
        const range = ownerDoc.createRange();
        range.selectNodeContents(el);
        range.collapse(false);
        sel.removeAllRanges();
        sel.addRange(range);
      }
    } catch(e) {}
    return true;
  }
  return false;
}

function setupSendResetListener() {
  function handleSendTrigger(e) {
    try {
      const ticketInfo = getActiveTicketDetails();
      const tState = getTicketState(ticketInfo.ticketKey);
      if (e.type === 'keydown' && e.key === 'Enter' && !e.shiftKey) {
        const comp = findZendeskComposer(ticketInfo.container);
        if (comp && (e.target === comp || comp.contains(e.target))) {
          tState.manualResetTime = Date.now();
          tState.msgId = '';
          tState.baseElapsedSec = 0;
          tState.anchorTime = Date.now();
          setTimeout(updateTimerDisplay, 80);
        }
      }
      if (e.type === 'click') {
        const btn = e.target.closest('[data-test-id*="send"], [data-test-id*="submit"], button[aria-label*="Send" i], [data-test-id="omni-composer-send-button"]');
        if (btn) {
          tState.manualResetTime = Date.now();
          tState.msgId = '';
          tState.baseElapsedSec = 0;
          tState.anchorTime = Date.now();
          setTimeout(updateTimerDisplay, 80);
        }
      }
    } catch(err) {}
  }
  document.addEventListener('keydown', handleSendTrigger, true);
  document.addEventListener('click', handleSendTrigger, true);
  return function cleanupSendResetListener() {
    document.removeEventListener('keydown', handleSendTrigger, true);
    document.removeEventListener('click', handleSendTrigger, true);
  };
}
